# CLAUDE.md — SliceUI: Image to Code Converter

## Project overview

SliceUI converts UI screenshots into clean, framework-specific frontend code.
Target users: frontend engineers who need a jump-start when slicing a new UI
feature into an existing project. Output is always a self-contained component —
never a full app scaffold.

---

## Tech stack

- **Framework**: Next.js 14 (App Router)
- **Styling**: Tailwind CSS
- **Language**: TypeScript strict
- **AI engine**: Gemini 2.0 Flash (free) + Groq LLaVA (free fallback)
- **Image processing**: sharp (normalize before sending)
- **Code display**: react-syntax-highlighter
- **Deploy**: Vercel

### Why free AI is enough

Gemini 2.0 Flash free tier:
- 1.500 requests/day, 15 RPM
- Supports vision (image input) AND text generation
- Quality comparable to paid vision models
- No credit card required — get key at aistudio.google.com

Groq free tier (automatic fallback when Gemini hits 429):
- 14.400 requests/day, 30 RPM
- Uses llava-v1.5-7b-4096-preview for vision
- Kicks in automatically — user never notices

---

## Pipeline — single-stage (vision + codegen in one call)

Unlike two-stage approaches, we send the image AND get code back in one request.
Gemini 2.0 Flash is multimodal — it can analyze the image AND generate
framework-specific code simultaneously. This is faster and simpler.

```
[Image + Framework choice]
        ↓
[Gemini 2.0 Flash — one call]
        ↓
[Component code — ready to paste]
```

Fallback chain (automatic, no user action needed):
```
Gemini → 429 rate limit? → Groq LLaVA → 429? → queue + retry message
```

## Core function

```typescript
// lib/ai.ts

import { GoogleGenerativeAI } from "@google/generative-ai"
import Groq from "groq-sdk"

const gemini = new GoogleGenerativeAI(process.env.GEMINI_API_KEY!)
const groq   = new Groq({ apiKey: process.env.GROQ_API_KEY })

export async function imageToCode(
  base64Image : string,
  framework   : Framework,
  options     : ConversionOptions
): Promise<string> {
  try {
    return await callGemini(base64Image, framework, options)
  } catch (err: any) {
    if (err?.status === 429 || err?.message?.includes("quota")) {
      console.warn("Gemini rate limit — switching to Groq")
      return await callGroq(base64Image, framework, options)
    }
    throw err
  }
}

// ── Gemini ────────────────────────────────────────────────────────────────────

async function callGemini(
  base64Image : string,
  framework   : Framework,
  options     : ConversionOptions
): Promise<string> {
  const model  = gemini.getGenerativeModel({ model: "gemini-2.0-flash" })
  const prompt = buildPrompt(framework, options)

  const result = await model.generateContent([
    prompt,
    { inlineData: { mimeType: "image/png", data: base64Image } }
  ])

  return clean(result.response.text())
}

// ── Groq ──────────────────────────────────────────────────────────────────────

async function callGroq(
  base64Image : string,
  framework   : Framework,
  options     : ConversionOptions
): Promise<string> {
  const prompt = buildPrompt(framework, options)

  const res = await groq.chat.completions.create({
    model: "meta-llama/llama-4-scout-17b-16e-instruct", // Groq vision model
    messages: [
      {
        role: "user",
        content: [
          { type: "text",      text: prompt },
          { type: "image_url", image_url: { url: `data:image/png;base64,${base64Image}` } }
        ]
      }
    ],
    max_tokens: 4096
  })

  return clean(res.choices[0].message.content ?? "")
}

// ── Helpers ───────────────────────────────────────────────────────────────────

function clean(raw: string): string {
  return raw
    .replace(/^```[\w-]*\n?/m, "")
    .replace(/```\s*$/m, "")
    .trim()
}
```

export function buildPrompt(framework: Framework, options: ConversionOptions): string {
  return `
You are a senior frontend engineer. Analyze this UI screenshot and generate
production-ready ${framework} component code.

CRITICAL OUTPUT RULES:
1. Output ONLY the component code. No explanation, no markdown fences.
   First character of your response must be the first character of the code.
2. Self-contained component — not a full app. No html/body/main entry points.
3. Use the exact text visible in the image as placeholder content.
4. For images in the UI: use a gray div with descriptive alt text.
5. For icons: use a comment /* icon: {description} */ as placeholder.
6. Match colors from the image exactly using hex values.
7. First line must be: // Generated by SliceUI

${options.responsive    ? "- Add responsive breakpoints (mobile-first)."                   : ""}
${options.semanticHtml  ? "- Use semantic HTML5 elements (nav, main, section, article)."   : ""}
${options.darkMode      ? "- Include dark mode variant."                                    : ""}
${options.a11y          ? "- Add aria-label, role, and alt attributes."                     : ""}

FRAMEWORK: ${framework}
${FRAMEWORK_RULES[framework]}
`
}

## API route

```typescript
// app/api/convert/route.ts
import { NextRequest, NextResponse } from "next/server"
import sharp from "sharp"
import { imageToCode } from "@/lib/ai"
import { validateImageFile } from "@/lib/imageUtils"

export const maxDuration = 60
export const runtime = "nodejs"

// Simple in-memory rate limiter (replace with Redis for production)
const rateMap = new Map<string, { count: number; reset: number }>()
const FREE_LIMIT = 5
const DAY_MS = 86_400_000

export async function POST(req: NextRequest) {
  // Rate limiting
  const ip = req.headers.get("x-forwarded-for") ?? "unknown"
  const now = Date.now()
  const entry = rateMap.get(ip)

  if (entry && now < entry.reset) {
    if (entry.count >= FREE_LIMIT) {
      return NextResponse.json(
        { error: "daily_limit_reached", message: "5 free conversions per day." },
        { status: 429 }
      )
    }
    entry.count++
  } else {
    rateMap.set(ip, { count: 1, reset: now + DAY_MS })
  }

  // Parse form
  const form      = await req.formData()
  const file      = form.get("image")    as File
  const framework = form.get("framework") as Framework
  const options   = JSON.parse(form.get("options") as string) as ConversionOptions

  // Validate
  const check = validateImageFile(file)
  if (!check.valid) {
    return NextResponse.json({ error: check.error }, { status: 400 })
  }

  // Normalize image: resize + convert to PNG
  const buf        = Buffer.from(await file.arrayBuffer())
  const normalized = await sharp(buf)
    .resize(1600, 1600, { fit: "inside", withoutEnlargement: true })
    .png()
    .toBuffer()
  const base64 = normalized.toString("base64")

  // Generate
  try {
    const code = await imageToCode(base64, framework, options)
    return NextResponse.json({ code })
  } catch (err: any) {
    console.error(err)
    return NextResponse.json(
      { error: "Generation failed", message: err.message },
      { status: 500 }
    )
  }
}
```
